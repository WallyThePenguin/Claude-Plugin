# RubricStore Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Deliver the produced quality rubric into the live Studio plugin — a compact rubric catalog always-on in the leading preamble + a `use_rubric(category)` tool that returns the full rubric on demand — so the agent builds TO the bar.

**Architecture:** Mirror the skills system exactly. A `RubricStore` (built-ins only) holds rubrics embedded as JSON strings (generated from the repo artifacts by a script), decoded + validated at construction. A pure `RubricCatalog` does validate/render/catalog/find (unit-testable via an in-Studio Spec). A parent-only `use_rubric` tool wraps the store. Wired into `deps`, the `AgentLoop` preamble, the Widget call, the relay's local-tools set, and `SystemPrompt`.

**Tech Stack:** Luau (Roblox Studio plugin, `--!strict`); Python 3 stdlib (the generator); rokit toolchain (`rojo`/`selene`/`stylua`). `HttpService:JSONDecode` for the embedded rubric JSON.

**Source spec:** `docs/superpowers/specs/2026-06-20-studio-agent-rubricstore-design.md`.

```python
# scripts/gen_builtin_rubrics.py
"""Generate src/BuiltinRubrics.luau from docs/superpowers/rubrics/*.rubric.json.

Each rubric is embedded as its (compact) JSON text inside a Luau long-bracket string at a
bracket level guaranteed absent from the text. RubricStore decodes + validates at load.
Do not hand-edit the output.
"""
import glob
import json
import os

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
SRC = glob.glob(os.path.join(ROOT, "docs", "superpowers", "rubrics", "*.rubric.json"))
OUT = os.path.join(ROOT, "src", "BuiltinRubrics.luau")


def bracket_level(text):
    """Smallest level n>=2 whose closing delimiter ]=...=] is absent from text."""
    n = 2
    while ("]" + "=" * n + "]") in text:
        n += 1
    return n


def main():
    payloads = []
    for path in sorted(SRC):
        with open(path, encoding="utf-8") as f:
            obj = json.load(f)  # validates it parses
        compact = json.dumps(obj, ensure_ascii=False, separators=(",", ":"))
        payloads.append((os.path.basename(path), compact))

    lines = [
        "--!strict",
        "-- BuiltinRubrics: GENERATED by scripts/gen_builtin_rubrics.py from",
        "-- docs/superpowers/rubrics/*.rubric.json. Do NOT hand-edit. Each entry is the raw rubric",
        "-- JSON; RubricStore decodes + validates it at load (HttpService:JSONDecode).",
        "",
        "local BuiltinRubrics = {}",
        "",
        "BuiltinRubrics.list = {",
    ]
    for name, compact in payloads:
        n = bracket_level(compact)
        op, cl = "[" + "=" * n + "[", "]" + "=" * n + "]"
        lines.append(f"\t-- {name}")
        lines.append(f"\t{op}{compact}{cl},")
    lines.append("}")
    lines.append("")
    lines.append("return BuiltinRubrics")
    lines.append("")

    with open(OUT, "w", encoding="utf-8", newline="\n") as f:
        f.write("\n".join(lines))

    # round-trip self-check: re-load each source and confirm the compact form re-parses
    ok = 0
    for _, compact in payloads:
        json.loads(compact)
        ok += 1
    print(f"generated {OUT} from {len(payloads)} rubric(s); round-trip OK: {ok}")


if __name__ == "__main__":
    main()
```

```lua
--!strict
-- RubricCatalog: PURE helpers for quality rubrics (no self, no DataModel — directly unit-testable;
-- mirrors SkillCatalog/Memory's pure layer). A rubric is the graded quality bar for an asset category;
-- the agent loads it via use_rubric and builds to the highest tier. validateRubric/render/catalogBlock/
-- findByCategory are shared by RubricStore + Tools/Rubrics + the Spec.

local RubricCatalog = {}

export type Grading = { pass: string, partial: string, fail: string }
export type Criterion = {
	id: string,
	order_index: number,
	tier: string, -- "baseline" | "good" | "exceptional"
	statement: string,
	above_generic_rationale: string,
	grading: Grading,
	evidence_required: string,
	concrete_fix: string,
	exemplar_refs: { string },
}
export type Rubric = {
	module: string,
	version: string,
	system_module: string,
	criteria: { Criterion },
}

local TIERS: { [string]: boolean } = { baseline = true, good = true, exceptional = true }
local TIER_ORDER = { "baseline", "good", "exceptional" }
local TIER_TITLE: { [string]: string } = {
	baseline = "Baseline",
	good = "Good (above the mean)",
	exceptional = "Exceptional (polish)",
}

local function trim(s: string): string
	return (string.gsub(s, "^%s*(.-)%s*$", "%1"))
end

-- Normalize a category name for matching against rubric.module: lowercase + trim; nil if empty.
function RubricCatalog.normalizeCategory(name: any): string?
	if type(name) ~= "string" then
		return nil
	end
	local s = string.lower(trim(name))
	if s == "" then
		return nil
	end
	return s
end

local function tierCounts(rubric: Rubric): { [string]: number }
	local n: { [string]: number } = { baseline = 0, good = 0, exceptional = 0 }
	for _, c in ipairs(rubric.criteria) do
		n[c.tier] = (n[c.tier] or 0) + 1
	end
	return n
end

-- Validate ONE decoded rubric table into a clean Rubric, or (nil, errorString). Defensive so a
-- malformed built-in is dropped at load (never crashes the plugin).
function RubricCatalog.validateRubric(raw: any): (Rubric?, string?)
	if type(raw) ~= "table" then
		return nil, "rubric must be a table"
	end
	if type(raw.module) ~= "string" or trim(raw.module) == "" then
		return nil, "rubric needs a non-empty 'module'"
	end
	if type(raw.criteria) ~= "table" or #raw.criteria == 0 then
		return nil, "rubric '" .. raw.module .. "' needs a non-empty 'criteria' array"
	end
	local criteria: { Criterion } = {}
	for i, c in ipairs(raw.criteria :: { any }) do
		if type(c) ~= "table" then
			return nil, "rubric '" .. raw.module .. "' criterion " .. tostring(i) .. " is not a table"
		end
		local g = c.grading
		if
			type(c.id) ~= "string"
			or type(c.tier) ~= "string"
			or not TIERS[c.tier]
			or type(c.statement) ~= "string"
			or type(g) ~= "table"
			or type(g.pass) ~= "string"
			or type(g.partial) ~= "string"
			or type(g.fail) ~= "string"
		then
			return nil,
				"rubric '" .. raw.module .. "' criterion " .. tostring(c.id) .. " is malformed"
		end
		local refs: { string } = {}
		if type(c.exemplar_refs) == "table" then
			for _, r in ipairs(c.exemplar_refs :: { any }) do
				if type(r) == "string" then
					table.insert(refs, r)
				end
			end
		end
		table.insert(criteria, {
			id = c.id,
			order_index = if type(c.order_index) == "number" then c.order_index else i,
			tier = c.tier,
			statement = c.statement,
			above_generic_rationale = if type(c.above_generic_rationale) == "string"
				then c.above_generic_rationale
				else "",
			grading = { pass = g.pass, partial = g.partial, fail = g.fail },
			evidence_required = if type(c.evidence_required) == "string"
				then c.evidence_required
				else "",
			concrete_fix = if type(c.concrete_fix) == "string" then c.concrete_fix else "",
			exemplar_refs = refs,
		})
	end
	table.sort(criteria, function(a, b)
		return a.order_index < b.order_index
	end)
	local moduleLower = string.lower(trim(raw.module))
	return {
		module = moduleLower,
		version = if type(raw.version) == "string" then raw.version else "1",
		system_module = if type(raw.system_module) == "string"
			then raw.system_module
			else moduleLower,
		criteria = criteria,
	},
		nil
end

-- Exact (normalized) category lookup against rubric.module.
function RubricCatalog.findByCategory(rubrics: { Rubric }, category: any): Rubric?
	local target = RubricCatalog.normalizeCategory(category)
	if target == nil then
		return nil
	end
	for _, r in ipairs(rubrics) do
		if r.module == target then
			return r
		end
	end
	return nil
end

-- Render the full rubric text use_rubric returns (tiered).
function RubricCatalog.render(rubric: Rubric): string
	local n = tierCounts(rubric)
	local out: { string } = {
		"# Quality rubric: " .. rubric.module .. " (v" .. rubric.version .. ")",
		"Build to the HIGHEST tier you can; for each criterion address its concrete_fix. Tiers: "
			.. tostring(n.baseline) .. " baseline / " .. tostring(n.good) .. " good / " .. tostring(
				n.exceptional
			) .. " exceptional.",
	}
	for _, tier in ipairs(TIER_ORDER) do
		local headerDone = false
		for _, c in ipairs(rubric.criteria) do
			if c.tier == tier then
				if not headerDone then
					table.insert(out, "")
					table.insert(out, "## " .. TIER_TITLE[tier])
					headerDone = true
				end
				table.insert(out, "")
				table.insert(out, "### " .. c.id)
				table.insert(out, c.statement)
				if c.above_generic_rationale ~= "" then
					table.insert(out, "- Why above generic: " .. c.above_generic_rationale)
				end
				table.insert(out, "- Pass: " .. c.grading.pass)
				table.insert(out, "- Partial: " .. c.grading.partial)
				table.insert(out, "- Fail: " .. c.grading.fail)
				if c.evidence_required ~= "" then
					table.insert(out, "- Evidence required: " .. c.evidence_required)
				end
				if c.concrete_fix ~= "" then
					table.insert(out, "- Concrete fix: " .. c.concrete_fix)
				end
				if #c.exemplar_refs > 0 then
					table.insert(out, "- Exemplar refs: " .. table.concat(c.exemplar_refs, ", "))
				end
			end
		end
	end
	return table.concat(out, "\n")
end

-- Always-on compact catalog (one line per rubric). "" if none.
function RubricCatalog.catalogBlock(rubrics: { Rubric }): string
	if #rubrics == 0 then
		return ""
	end
	local out: { string } = {
		"# Quality rubrics (build to the bar)",
		"Curated above-the-mean quality rubrics per asset category. BEFORE building an asset in a "
			.. "listed category, call use_rubric(category) to load its full tiered rubric, then build to "
			.. "the highest tier you can and address each criterion's concrete_fix.",
		"",
	}
	for _, r in ipairs(rubrics) do
		local n = tierCounts(r)
		table.insert(
			out,
			"- " .. r.module .. " — " .. tostring(#r.criteria) .. " criteria (" .. tostring(n.baseline)
				.. " baseline / " .. tostring(n.good) .. " good / " .. tostring(
					n.exceptional
				) .. " exceptional)"
		)
	end
	return table.concat(out, "\n")
end

return RubricCatalog
```

```lua
--!strict
-- RubricCatalogSpec: in-Studio self-test for the PURE RubricCatalog helpers (no DataModel). Run via
-- temporarily adding `require(script.RubricCatalogSpec).run()` to init.server.luau, build, reopen
-- Studio, read Output. Mirrors SkillCatalogSpec.

local RubricCatalog = require(script.Parent.RubricCatalog)

local M = {}

type Case = { name: string, fn: () -> (boolean, string?) }

local function goodRaw(): any
	return {
		module = "Locomotion",
		version = "1.1",
		criteria = {
			{
				id = "c-good",
				order_index = 2,
				tier = "good",
				statement = "Warps by speed AND direction.",
				above_generic_rationale = "Default plays one loop.",
				grading = { pass = "blend tree", partial = "speed only", fail = "single loop" },
				evidence_required = "strafe shows side motion",
				concrete_fix = "add a 2D blend space",
				exemplar_refs = { "https://example.com/x" },
			},
			{
				id = "c-base",
				order_index = 1,
				tier = "baseline",
				statement = "Idle/walk/run exist.",
				grading = { pass = "p", partial = "pa", fail = "f" },
			},
		},
	}
end

local function buildCases(): { Case }
	local cases: { Case } = {}
	local function add(name: string, fn: () -> (boolean, string?))
		table.insert(cases, { name = name, fn = fn })
	end

	add("validateRubric accepts a good rubric, lowercases module, sorts by order_index", function()
		local r, err = RubricCatalog.validateRubric(goodRaw())
		return err == nil
			and r ~= nil
			and r.module == "locomotion"
			and r.system_module == "locomotion"
			and #r.criteria == 2
			and r.criteria[1].id == "c-base" -- order_index 1 first
			and r.criteria[2].id == "c-good"
	end)

	add("validateRubric rejects bad inputs", function()
		local _, e1 = RubricCatalog.validateRubric("nope")
		local _, e2 = RubricCatalog.validateRubric({ module = "x", criteria = {} })
		local bad = goodRaw()
		bad.criteria[1].tier = "amazing"
		local _, e3 = RubricCatalog.validateRubric(bad)
		local bad2 = goodRaw()
		bad2.criteria[1].grading = { pass = "p" } -- missing partial/fail
		local _, e4 = RubricCatalog.validateRubric(bad2)
		return e1 ~= nil and e2 ~= nil and e3 ~= nil and e4 ~= nil
	end)

	add("render contains tiered headers + a concrete_fix + the version", function()
		local r = (RubricCatalog.validateRubric(goodRaw()))
		assert(r ~= nil)
		local text = RubricCatalog.render(r :: any)
		return string.find(text, "v1.1", 1, true) ~= nil
			and string.find(text, "## Baseline", 1, true) ~= nil
			and string.find(text, "## Good", 1, true) ~= nil
			and string.find(text, "add a 2D blend space", 1, true) ~= nil
	end)

	add("catalogBlock lists a line per rubric with counts; empty -> ''", function()
		local r = (RubricCatalog.validateRubric(goodRaw()))
		assert(r ~= nil)
		local block = RubricCatalog.catalogBlock({ r :: any })
		return RubricCatalog.catalogBlock({}) == ""
			and string.find(block, "locomotion", 1, true) ~= nil
			and string.find(block, "2 criteria", 1, true) ~= nil
			and string.find(block, "1 baseline", 1, true) ~= nil
	end)

	add("findByCategory matches case-insensitively; miss + bad input -> nil", function()
		local r = (RubricCatalog.validateRubric(goodRaw()))
		assert(r ~= nil)
		local list = { r :: any }
		return RubricCatalog.findByCategory(list, "LOCOMOTION") ~= nil
			and RubricCatalog.findByCategory(list, "map") == nil
			and RubricCatalog.findByCategory(list, 5) == nil
	end)

	return cases
end

function M.run(): boolean
	local cases = buildCases()
	local passed = 0
	for _, c in ipairs(cases) do
		local ok, err = c.fn()
		if ok then
			passed += 1
			print("[RubricCatalogSpec] PASS " .. c.name)
		else
			warn("[RubricCatalogSpec] FAIL " .. c.name .. (if err then " — " .. err else ""))
		end
	end
	print(("[RubricCatalogSpec] %d/%d passed"):format(passed, #cases))
	return passed == #cases
end

return M
```

```lua
--!strict
-- RubricStore: the live rubric store. Holds BUILT-IN rubrics (produced by the train-category pipeline,
-- embedded as JSON strings in BuiltinRubrics — generated by scripts/gen_builtin_rubrics.py). Decodes +
-- validates them ONCE at construction; a malformed one is dropped + warned (never crashes). One
-- instance in deps.rubrics. catalogBlock() rides the leading preamble (always-on); resolve(category)
-- backs the use_rubric tool (full rubric on demand). Built-ins only in v1 (no user rubrics / save).

local HttpService = game:GetService("HttpService")

local RubricCatalog = require(script.Parent.RubricCatalog)
local BuiltinRubrics = require(script.Parent.BuiltinRubrics)

type Rubric = RubricCatalog.Rubric

local RubricStore = {}
RubricStore.__index = RubricStore

export type RubricStore = typeof(setmetatable({} :: { _rubrics: { Rubric } }, RubricStore))

-- Decode + validate each embedded JSON string ONCE; drop + warn on a malformed/unparseable one.
local function buildBuiltins(): { Rubric }
	local out: { Rubric } = {}
	local seen: { [string]: boolean } = {}
	for _, jsonText in ipairs(BuiltinRubrics.list :: { any }) do
		if type(jsonText) == "string" then
			local ok, decoded = pcall(function()
				return HttpService:JSONDecode(jsonText)
			end)
			if ok then
				local rubric, err = RubricCatalog.validateRubric(decoded)
				if rubric ~= nil and not seen[rubric.module] then
					seen[rubric.module] = true
					table.insert(out, rubric)
				elseif rubric == nil then
					warn("[StudioAgent] skipping malformed built-in rubric: " .. tostring(err))
				end
			else
				warn("[StudioAgent] skipping unparseable built-in rubric JSON")
			end
		end
	end
	return out
end

function RubricStore.new(): RubricStore
	return setmetatable({ _rubrics = buildBuiltins() }, RubricStore)
end

function RubricStore.all(self: RubricStore): { Rubric }
	return self._rubrics
end

function RubricStore.listBuiltins(self: RubricStore): { Rubric }
	return self._rubrics
end

-- The always-on catalog injected into the leading preamble.
function RubricStore.catalogBlock(self: RubricStore): string
	return RubricCatalog.catalogBlock(self._rubrics)
end

-- Resolve a category into the full rubric text use_rubric returns. (text, ok).
function RubricStore.resolve(self: RubricStore, category: any): (string, boolean)
	local rubric = RubricCatalog.findByCategory(self._rubrics, category)
	if rubric == nil then
		local names: { string } = {}
		for _, r in ipairs(self._rubrics) do
			table.insert(names, r.module)
		end
		local avail = if #names > 0 then table.concat(names, ", ") else "(none)"
		return "unknown rubric '" .. tostring(category) .. "'. Available: " .. avail, false
	end
	return RubricCatalog.render(rubric), true
end

return RubricStore
```

```lua
--!strict
-- Rubric tools: use_rubric. Thin wrapper over context.rubrics (the RubricStore). Parent-only: NOT in
-- the subagent/writer allowlists, and a child's context carries no rubrics, so it no-ops gracefully for
-- a child. Non-destructive (no DataModel writes).

local Rubrics = {}

export type Descriptor = {
	name: string,
	description: string,
	inputSchema: { [string]: any },
	destructive: boolean,
	run: (input: any, context: any) -> (string, boolean),
}

local function store(context: any): any
	return if type(context) == "table" then context.rubrics else nil
end

local function useRubric(input: any, context: any): (string, boolean)
	local s = store(context)
	if s == nil then
		return "use_rubric is not available here (no rubrics in this context)", true
	end
	local args = if type(input) == "table" then input else {}
	if type(args.category) ~= "string" then
		return "use_rubric requires { category = <string> }", true
	end
	local text, ok = s:resolve(args.category)
	return text, not ok
end

Rubrics.descriptors = {
	{
		name = "use_rubric",
		description = "Load the quality RUBRIC for an asset category and build TO it. Rubrics are listed "
			.. "in your context under '# Quality rubrics'. This returns the full tiered rubric "
			.. "(baseline/good/exceptional criteria with evidence + concrete fixes) — read it, then build "
			.. "to the highest tier you can, addressing each criterion. Call this BEFORE building an asset "
			.. "in a listed category.",
		inputSchema = {
			type = "object",
			properties = {
				category = {
					type = "string",
					description = "The rubric category from the catalog (e.g. 'locomotion').",
				},
			},
			required = { "category" },
		},
		destructive = false,
		run = useRubric,
	},
} :: { Descriptor }

return Rubrics
```
