```lua
local BuildGuideCatalog = require(script.Parent.BuildGuideCatalog)

local M = {}

type Case = { name: string, fn: () -> (boolean, string?) }

local function goodRaw(): any
	return {
		module = "Locomotion",
		version = "1.0",
		rubric_version = "1.0",
		overview = "Build a velocity-driven blend system.",
		killer_systems = { { name = "blend tree", what = "speed/dir warp", how = "2D blend space" } },

			{
				tier = "good",
				goal = "above the mean",
				steps = {
					{
						id = "loco-blend",
						order_index = 2,
						instruction = "Author a 2D blend space across speed and direction.",
						addresses_criteria = { "loco-blend-tree" },
						tools = { "add_keyframe", "bind_animation" },
						pitfalls = "don't key one loop",
					},
				},
			},
			{
				tier = "baseline",
				goal = "it moves",
				steps = {
					{ id = "loco-loops", order_index = 1, instruction = "Make idle/walk/run loops." },
				},
			},
		},
		provenance = { coverage = { criteria_total = 2, criteria_addressed = 2, uncovered = {} } },
	}
end

local function buildCases(): { Case }
	local cases: { Case } = {}
	local function add(name: string, fn: () -> (boolean, string?))
		table.insert(cases, { name = name, fn = fn })
	end

	add("validateGuide accepts a good guide, lowercases module, sorts tiers + steps", function()
		local g, err = BuildGuideCatalog.validateGuide(goodRaw())
		return err == nil
			and g ~= nil
			and g.module == "locomotion"
			and g.system_module == "locomotion"
			and #g.tiers == 2
			and g.tiers[1].tier == "baseline" -- tier order, not input order
			and g.tiers[2].tier == "good"
			and g.tiers[2].steps[1].id == "loco-blend"
	end)

	add("validateGuide rejects bad inputs", function()
		local _, e1 = BuildGuideCatalog.validateGuide("nope")
		local _, e2 = BuildGuideCatalog.validateGuide({ module = "x", tiers = {} })
		local bad = goodRaw()
		bad.tiers[1].tier = "amazing"
		local _, e3 = BuildGuideCatalog.validateGuide(bad)
		local bad2 = goodRaw()
		bad2.tiers[2].steps[1].id = nil -- step missing id
		local _, e4 = BuildGuideCatalog.validateGuide(bad2)
		return e1 ~= nil and e2 ~= nil and e3 ~= nil and e4 ~= nil
	end)

	add("render contains tiered headers + an instruction + tool tags + the version", function()
		local g = (BuildGuideCatalog.validateGuide(goodRaw()))
		assert(g ~= nil, "validateGuide returned nil for a good guide")
		local text = BuildGuideCatalog.render(g :: any)
		return string.find(text, "v1.0", 1, true) ~= nil
			and string.find(text, "## Baseline", 1, true) ~= nil
			and string.find(text, "## Good", 1, true) ~= nil
			and string.find(text, "2D blend space", 1, true) ~= nil
			and string.find(text, "add_keyframe", 1, true) ~= nil
			and string.find(text, "loco-blend-tree", 1, true) ~= nil
	end)

	add("catalogBlock lists a line per guide with counts + coverage; empty -> ''", function()
		local g = (BuildGuideCatalog.validateGuide(goodRaw()))
		assert(g ~= nil, "validateGuide returned nil for a good guide")
		local block = BuildGuideCatalog.catalogBlock({ g :: any })
		return BuildGuideCatalog.catalogBlock({}) == ""
			and string.find(block, "locomotion", 1, true) ~= nil
			and string.find(block, "2 steps", 1, true) ~= nil
			and string.find(block, "2/2", 1, true) ~= nil
	end)

	add("findByCategory matches case-insensitively; miss + bad input -> nil", function()
		local g = (BuildGuideCatalog.validateGuide(goodRaw()))
		assert(g ~= nil, "validateGuide returned nil for a good guide")
		local list = { g :: any }
		return BuildGuideCatalog.findByCategory(list, "LOCOMOTION") ~= nil
			and BuildGuideCatalog.findByCategory(list, "map") == nil
			and BuildGuideCatalog.findByCategory(list, 5) == nil
	end)

	return cases
end

function M.run(): boolean
	local cases = buildCases()
	local passed = 0
	for _, c in ipairs(cases) do
		local ok, err = c.fn()
		if ok then
			passed += 1
			print("[BuildGuideCatalogSpec] PASS " .. c.name)
		else
			warn("[BuildGuideCatalogSpec] FAIL " .. c.name .. (if err then " — " .. err else ""))
		end
	end
	print(("[BuildGuideCatalogSpec] %d/%d passed"):format(passed, #cases))
	return passed == #cases
end

return M
```

```lua
--!strict
-- BuildGuideCatalog: PURE helpers for build guides (no self, no DataModel — directly unit-testable;
-- mirrors RubricCatalog's pure layer). A build guide is the proactive twin of a rubric: the rubric
-- GRADES, the guide BUILDS. The agent loads it via use_build_guide and follows the tiered steps.
-- validateGuide/render/catalogBlock/findByCategory are shared by BuildGuideStore + Tools/BuildGuides + Spec.

local BuildGuideCatalog = {}

export type Step = {
	id: string,
	order_index: number,
	instruction: string,
	addresses_criteria: { string },
	tools: { string },
	pitfalls: string,
}
export type Tier = {
	tier: string, -- "baseline" | "good" | "exceptional"
	goal: string,
	steps: { Step },
}
export type KillerSystem = { name: string, what: string, how: string }
export type Coverage = { total: number, addressed: number }
export type Guide = {
	module: string,
	version: string,
	system_module: string,
	rubric_version: string,
	overview: string,
	killer_systems: { KillerSystem },
	tiers: { Tier },
	common_pitfalls: { string },
	coverage: Coverage?,
}

local TIERS: { [string]: boolean } = { baseline = true, good = true, exceptional = true }
local TIER_ORDER = { "baseline", "good", "exceptional" }
local TIER_RANK: { [string]: number } = { baseline = 1, good = 2, exceptional = 3 }
local TIER_TITLE: { [string]: string } = {
	baseline = "Baseline",
	good = "Good (above the mean)",
	exceptional = "Exceptional (polish)",
}

local function trim(s: string): string
	return (string.gsub(s, "^%s*(.-)%s*$", "%1"))
end

-- Normalize a category name for matching against guide.module: lowercase + trim; nil if empty.
function BuildGuideCatalog.normalizeCategory(name: any): string?
	if type(name) ~= "string" then
		return nil
	end
	local s = string.lower(trim(name))
	if s == "" then
		return nil
	end
	return s
end

local function strArray(raw: any): { string }
	local out: { string } = {}
	if type(raw) == "table" then
		for _, v in ipairs(raw :: { any }) do
			if type(v) == "string" then
				table.insert(out, v)
			end
		end
	end
	return out
end

local function str(raw: any): string
	return if type(raw) == "string" then raw else ""
end

-- Validate ONE decoded guide table into a clean Guide, or (nil, errorString). Defensive so a malformed
-- built-in is dropped at load (never crashes the plugin). Mirrors RubricCatalog.validateRubric.
function BuildGuideCatalog.validateGuide(raw: any): (Guide?, string?)
	if type(raw) ~= "table" then
		return nil, "build guide must be a table"
	end
	if type(raw.module) ~= "string" or trim(raw.module) == "" then
		return nil, "build guide needs a non-empty 'module'"
	end
	if type(raw.tiers) ~= "table" or #raw.tiers == 0 then
		return nil, "build guide '" .. raw.module .. "' needs a non-empty 'tiers' array"
	end
	local tiers: { Tier } = {}
	for ti, t in ipairs(raw.tiers :: { any }) do
		if type(t) ~= "table" or type(t.tier) ~= "string" or not TIERS[t.tier] then
			return nil, "build guide '" .. raw.module .. "' tier " .. tostring(ti) .. " is malformed"
		end
		if type(t.steps) ~= "table" then
			return nil, "build guide '" .. raw.module .. "' tier '" .. t.tier .. "' needs a 'steps' array"
		end
		local steps: { Step } = {}
		for si, s in ipairs(t.steps :: { any }) do
			if type(s) ~= "table" or type(s.id) ~= "string" or type(s.instruction) ~= "string" then
				return nil,
					"build guide '" .. raw.module .. "' tier '" .. t.tier .. "' step " .. tostring(si)
						.. " needs an id + instruction"
			end
			table.insert(steps, {
				id = s.id,
				order_index = if type(s.order_index) == "number" then s.order_index else si,
				instruction = s.instruction,
				addresses_criteria = strArray(s.addresses_criteria),
				tools = strArray(s.tools),
				pitfalls = str(s.pitfalls),
			})
		end
		table.sort(steps, function(a, b)
			return a.order_index < b.order_index
		end)
		table.insert(tiers, { tier = t.tier, goal = str(t.goal), steps = steps })
	end
	table.sort(tiers, function(a, b)
		return (TIER_RANK[a.tier] or 9) < (TIER_RANK[b.tier] or 9)
	end)
	local killers: { KillerSystem } = {}
	if type(raw.killer_systems) == "table" then
		for _, k in ipairs(raw.killer_systems :: { any }) do
			if type(k) == "table" and type(k.name) == "string" then
				table.insert(killers, { name = k.name, what = str(k.what), how = str(k.how) })
			end
		end
	end
	local coverage: Coverage? = nil
	if type(raw.provenance) == "table" and type(raw.provenance.coverage) == "table" then
		local c = raw.provenance.coverage
		if type(c.criteria_total) == "number" and type(c.criteria_addressed) == "number" then
			coverage = { total = c.criteria_total, addressed = c.criteria_addressed }
		end
	end
	local moduleLower = string.lower(trim(raw.module))
	return {
		module = moduleLower,
		version = if type(raw.version) == "string" then raw.version else "1",
		system_module = if type(raw.system_module) == "string" then raw.system_module else moduleLower,
		rubric_version = str(raw.rubric_version),
		overview = str(raw.overview),
		killer_systems = killers,
		tiers = tiers,
		common_pitfalls = strArray(raw.common_pitfalls),
		coverage = coverage,
	},
		nil
end

-- Exact (normalized) category lookup against guide.module.
function BuildGuideCatalog.findByCategory(guides: { Guide }, category: any): Guide?
	local target = BuildGuideCatalog.normalizeCategory(category)
	if target == nil then
		return nil
	end
	for _, g in ipairs(guides) do
		if g.module == target then
			return g
		end
	end
	return nil
end

local function stepCount(guide: Guide): number
	local n = 0
	for _, t in ipairs(guide.tiers) do
		n += #t.steps
	end
	return n
end

-- Render the full guide text use_build_guide returns (tiered, ordered steps).
function BuildGuideCatalog.render(guide: Guide): string
	local header = "# Build guide: " .. guide.module .. " (v" .. guide.version .. ")"
	if guide.rubric_version ~= "" then
		header ..= " — targets the " .. guide.module .. " rubric (v" .. guide.rubric_version .. ")"
	end
	local out: { string } = { header }
	if guide.overview ~= "" then
		table.insert(out, guide.overview)
	end
	if #guide.killer_systems > 0 then
		local parts: { string } = {}
		for _, k in ipairs(guide.killer_systems) do
			local p = k.name
			if k.what ~= "" then
				p ..= " — " .. k.what
			end
			if k.how ~= "" then
				p ..= " (" .. k.how .. ")"
			end
			table.insert(parts, p)
		end
		table.insert(out, "Killer systems: " .. table.concat(parts, "; "))
	end
	for _, tier in ipairs(TIER_ORDER) do
		for _, t in ipairs(guide.tiers) do
			if t.tier == tier then
				table.insert(out, "")
				local h = "## " .. TIER_TITLE[tier]
				if t.goal ~= "" then
					h ..= " — " .. t.goal
				end
				table.insert(out, h)
				for i, s in ipairs(t.steps) do
					local line = tostring(i) .. ". " .. s.instruction
					if #s.addresses_criteria > 0 then
						line ..= "  [addresses: " .. table.concat(s.addresses_criteria, ", ") .. "]"
					end
					if #s.tools > 0 then
						line ..= "  [tools: " .. table.concat(s.tools, ", ") .. "]"
					end
					table.insert(out, line)
					if s.pitfalls ~= "" then
						table.insert(out, "   Pitfall: " .. s.pitfalls)
					end
				end
			end
		end
	end
	if #guide.common_pitfalls > 0 then
		table.insert(out, "")
		table.insert(out, "Common pitfalls: " .. table.concat(guide.common_pitfalls, "; "))
	end
	if guide.coverage ~= nil then
		table.insert(
			out,
			"Coverage: "
				.. tostring(guide.coverage.addressed)
				.. "/"
				.. tostring(guide.coverage.total)
				.. " rubric criteria addressed."
		)
	end
	return table.concat(out, "\n")
end

-- Always-on compact catalog (one line per guide). "" if none.
function BuildGuideCatalog.catalogBlock(guides: { Guide }): string
	if #guides == 0 then
		return ""
	end
	local out: { string } = {
		"# Build guides (how to build to the bar)",
		"For an asset category that has a rubric, call use_build_guide(category) to load its tiered BUILD "
			.. "approach — the steps (each citing real tools + the rubric criteria they satisfy) that take a "
			.. "build from baseline to exceptional. Pair it with use_rubric(category): the rubric is the bar, "
			.. "the build guide is how to reach it.",
		"",
	}
	for _, g in ipairs(guides) do
		local line = "- " .. g.module .. " — " .. tostring(stepCount(g)) .. " steps across "
			.. tostring(#g.tiers) .. " tiers"
		if g.coverage ~= nil then
			line ..= " (covers " .. tostring(g.coverage.addressed) .. "/" .. tostring(g.coverage.total)
				.. " rubric criteria)"
		end
		table.insert(out, line)
	end
	return table.concat(out, "\n")
end

return BuildGuideCatalog
```

```python
"""Generate src/BuiltinBuildGuides.luau from docs/superpowers/guides/*.guide.json.

Each guide is embedded as its (compact) JSON text inside a Luau long-bracket string at a
bracket level guaranteed absent from the text. BuildGuideStore decodes + validates at load.
Do not hand-edit the output. Mirrors gen_builtin_rubrics.py.
"""
import glob
import json
import os

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
SRC = glob.glob(os.path.join(ROOT, "docs", "superpowers", "guides", "*.guide.json"))
OUT = os.path.join(ROOT, "src", "BuiltinBuildGuides.luau")


def bracket_level(text):
    """Smallest level n>=2 whose closing delimiter ]=...=] is absent from text."""
    n = 2
    while ("]" + "=" * n + "]") in text:
        n += 1
    return n


def main():
    payloads = []
    for path in sorted(SRC):
        with open(path, encoding="utf-8") as f:
            obj = json.load(f)  # validates it parses
        compact = json.dumps(obj, ensure_ascii=False, separators=(",", ":"))
        payloads.append((os.path.basename(path), compact))

    lines = [
        "--!strict",
        "-- BuiltinBuildGuides: GENERATED by scripts/gen_builtin_buildguides.py from",
        "-- docs/superpowers/guides/*.guide.json. Do NOT hand-edit. Each entry is the raw guide",
        "-- JSON; BuildGuideStore decodes + validates it at load (HttpService:JSONDecode).",
        "",
        "local BuiltinBuildGuides = {}",
        "",
        "BuiltinBuildGuides.list = {",
    ]
    for name, compact in payloads:
        n = bracket_level(compact)
        op, cl = "[" + "=" * n + "[", "]" + "=" * n + "]"
        lines.append(f"\t-- {name}")
        lines.append(f"\t{op}{compact}{cl},")
```

```lua
--!strict
-- BuildGuideStore: the live build-guide store. Holds BUILT-IN guides (produced by the train-recipe
-- factory, embedded as JSON strings in BuiltinBuildGuides — generated by gen_builtin_buildguides.py).
-- Decodes + validates them ONCE at construction; a malformed one is dropped + warned (never crashes).
-- One instance in deps.buildGuides. catalogBlock() rides the leading preamble (always-on);
-- resolve(category) backs the use_build_guide tool. Built-ins only in v1 (no user guides / save).
-- Mirrors RubricStore.

local HttpService = game:GetService("HttpService")

local BuildGuideCatalog = require(script.Parent.BuildGuideCatalog)
local BuiltinBuildGuides = require(script.Parent.BuiltinBuildGuides)

type Guide = BuildGuideCatalog.Guide

local BuildGuideStore = {}
BuildGuideStore.__index = BuildGuideStore

export type BuildGuideStore = typeof(setmetatable({} :: { _guides: { Guide } }, BuildGuideStore))

local function buildBuiltins(): { Guide }
	local out: { Guide } = {}
	local seen: { [string]: boolean } = {}
	for _, jsonText in ipairs(BuiltinBuildGuides.list :: { any }) do
		if type(jsonText) == "string" then
			local ok, decoded = pcall(function()
				return HttpService:JSONDecode(jsonText)
			end)
			if ok then
				local guide, err = BuildGuideCatalog.validateGuide(decoded)
				if guide ~= nil and not seen[guide.module] then
					seen[guide.module] = true
					table.insert(out, guide)
				elseif guide == nil then
					warn("[StudioAgent] skipping malformed built-in build guide: " .. tostring(err))
				end
			else
				warn("[StudioAgent] skipping unparseable built-in build guide JSON")
			end
		end
	end
	return out
end

function BuildGuideStore.new(): BuildGuideStore
	return setmetatable({ _guides = buildBuiltins() }, BuildGuideStore)
end

function BuildGuideStore.all(self: BuildGuideStore): { Guide }
	return self._guides
end

-- The always-on catalog injected into the leading preamble.
function BuildGuideStore.catalogBlock(self: BuildGuideStore): string
	return BuildGuideCatalog.catalogBlock(self._guides)
end

-- Resolve a category into the full guide text use_build_guide returns. (text, ok).
function BuildGuideStore.resolve(self: BuildGuideStore, category: any): (string, boolean)
	local guide = BuildGuideCatalog.findByCategory(self._guides, category)
	if guide == nil then
		local names: { string } = {}
		for _, g in ipairs(self._guides) do
			table.insert(names, g.module)
		end
		local avail = if #names > 0 then table.concat(names, ", ") else "(none)"
		return "unknown build guide '" .. tostring(category) .. "'. Available: " .. avail, false
	end
	return BuildGuideCatalog.render(guide), true
end

return BuildGuideStore
```

```javascript
export const meta = {
  name: 'train-recipe',
  description: 'Build a rubric-anchored, above-the-mean BUILD GUIDE for one asset category (build-angled sweep -> distill into tiered steps -> above-generic step critic -> coverage fill vs the rubric -> assemble -> completeness)',
  phases: [
    { title: 'Sweep' },
    { title: 'Distill' },
    { title: 'Critique-steps' },
    { title: 'Cover' },
    { title: 'Assemble' },
    { title: 'Review' },
  ],

if (typeof cfg === 'string') { try { cfg = JSON.parse(cfg) } catch (e) { cfg = {} } }

const RUBRIC = cfg.rubric || { module: CATEGORY, criteria: [] }
const TOOLS = cfg.toolCatalog || []
const MAX_FILL = cfg.maxFillRounds || 3

function coveredIds(tiers) {
  const set = new Set()
  for (const t of (tiers || [])) for (const s of (t.steps || [])) for (const id of (s.addresses_criteria || [])) set.add(id)
  return set
}
function uncovered(tiers) {
  const cov = coveredIds(tiers)
  return CRIT_IDS.filter(id => !cov.has(id))
}
function allSteps(tiers) {
  const out = []
  for (const t of (tiers || [])) for (const s of (t.steps || [])) out.push({ ...s, tier: t.tier })

// ---- schemas ----
const FINDINGS_SCHEMA = { type: 'object', required: ['techniques'], properties: {
  angle: { type: 'string' },
  techniques: { type: 'array', items: { type: 'object', required: ['technique'], properties: {

    how_to_build: { type: 'string' }, tools: { type: 'array', items: { type: 'string' } },
    pitfall: { type: 'string' }, source: { type: 'string' } } } } } }

  id: { type: 'string' }, instruction: { type: 'string' },

  overview: { type: 'string' },
  killer_systems: { type: 'array', items: { type: 'object', required: ['name'], properties: {

  common_pitfalls: { type: 'array', items: { type: 'string' } },

    tier: { type: 'string' }, goal: { type: 'string' },
    steps: { type: 'array', items: STEP_OBJ } } } } } }

  id: { type: 'string' }, verdict: { type: 'string' },

  steps: { type: 'array', items: { type: 'object', required: ['id', 'instruction', 'tier', 'addresses_criteria'], properties: {
    id: { type: 'string' }, instruction: { type: 'string' }, tier: { type: 'string' },

  rubric_version: { type: 'string' }, overview: { type: 'string' },
  killer_systems: { type: 'array', items: { type: 'object', properties: {
    name: { type: 'string' }, what: { type: 'string' }, how: { type: 'string' } } } },

      addresses_criteria: { type: 'array', items: { type: 'string' } },
      tools: { type: 'array', items: { type: 'string' } }, pitfalls: { type: 'string' } } } } } } },
  common_pitfalls: { type: 'array', items: { type: 'string' } } } }

const REVIEW_SCHEMA = { type: 'object', required: ['assessment'], properties: {
  killer_systems_covered: { type: 'array', items: { type: 'string' } },
  killer_systems_missing: { type: 'array', items: { type: 'string' } },
  build_order_sound: { type: 'boolean' }, assessment: { type: 'string' } } }

// ---- Sweep: build-angled (how to BUILD, not what separates good/generic — the rubric has that) ----
phase('Sweep')
const ANGLES = [
  { key: 'workflow', prompt: `Search the Roblox creator-docs GitHub mirror + create.roblox.com + DevForum for the step-by-step WORKFLOW pros use to BUILD ${DESCRIPTION} (the build ORDER, what to set up first, how the pieces connect). Extract concrete BUILD techniques, each with a tier_hint (baseline/good/exceptional), how_to_build (the actual steps), the Roblox tools/APIs it uses, a pitfall, and the source URL.` },
  { key: 'killer-impl', prompt: `For ${DESCRIPTION}, research HOW to IMPLEMENT its killer systems in Roblox: ${JSON.stringify(KILLER)}. For each, find the concrete build technique (APIs, instances, structure). Extract techniques with tier_hint, how_to_build, tools, pitfall, source URL.` },
  { key: 'pitfalls', prompt: `Search DevForum CRITIQUE + "why does my ${CATEGORY} look bad / feel off" threads. Extract the BUILD MISTAKE and the FIX for each (the fix is the technique), with tier_hint, how_to_build (the corrected approach), tools, the pitfall it avoids, and source URL.` },
  { key: 'video', prompt: `Find YouTube build tutorials/breakdowns on ${DESCRIPTION} and read their transcripts. Extract the BUILD steps the author actually performs (in order), each with tier_hint, how_to_build, tools, pitfall, and source URL.` },

  agent(a.prompt, { label: `sweep:${a.key}`, phase: 'Sweep', schema: FINDINGS_SCHEMA })

phase('Distill')

  agent(`ABOVE-GENERIC BUILD-STEP JUDGE (STRICT). The GENERIC FLOOR for ${CATEGORY} is what default/generic builds + a basic tutorial already produce. for (const v of judged) verdictById[v.id] = v
for (const t of draft.tiers) {
  t.steps = (t.steps || []).filter(s => (verdictById[s.id]?.verdict !== 'reject')).map(s => {
    const v = verdictById[s.id]
    return (v && v.verdict === 'raise' && v.raised_instruction) ? { ...s, instruction: v.raised_instruction } : s
  })
}
log(`Critique: ${steps.length - allSteps(draft.tiers).length} generic steps dropped; uncovered now: ${uncovered(draft.tiers).length}`)

// ---- Cover: fill loop until every rubric criterion is addressed by some step ----
phase('Cover')
let round = 0
while (uncovered(draft.tiers).length > 0 && round < MAX_FILL) {
  const gaps = uncovered(draft.tiers)
  const fill = await agent(
    `These rubric criteria are NOT yet addressed by any build step in the "${CATEGORY}" guide. Write the concrete BUILD step(s) that would satisfy each — above the generic floor, citing the real tools, assigned to the right tier (match the criterion's tier). If a criterion is genuinely un-buildable with the available tools, still return a best-effort step describing what WOULD satisfy it and note the gap in pitfalls.\n\nUNADDRESSED CRITERIA:\n${JSON.stringify((RUBRIC.criteria || []).filter(c => gaps.includes(c.id)), null, 1)}\n\nREAL TOOLS: ${JSON.stringify(TOOLS)}`,
    { label: `fill#${round}`, phase: 'Cover', schema: FILL_SCHEMA })
  for (const ns of (fill.steps || [])) {
    let tier = draft.tiers.find(t => t.tier === ns.tier)
    if (!tier) { tier = { tier: ns.tier || 'good', goal: '', steps: [] }; draft.tiers.push(tier) }
    tier.steps.push({ id: ns.id, instruction: ns.instruction, addresses_criteria: ns.addresses_criteria || [], tools: ns.tools || [], pitfalls: ns.pitfalls || '' })
  }
  round++
  log(`Cover r${round}: uncovered ${uncovered(draft.tiers).length}/${CRIT_IDS.length}`)
}
const finalUncovered = uncovered(draft.tiers)

// ---- Assemble: to the build-guide schema ----
phase('Assemble')
const assembled = await agent(
  `Assemble this into a FINAL build guide for category "${CATEGORY}" following the schema EXACTLY. Set module="${CATEGORY}", system_module="${CATEGORY}", version="1.0", rubric_version="${RUBRIC.version || '1.0'}". Order tiers baseline -> good -> exceptional; within each tier assign a stable order_index in build order starting at 1. Keep every step's addresses_criteria + tools + pitfalls. Preserve overview, killer_systems, common_pitfalls.\n\nDRAFT:\n${JSON.stringify(draft, null, 1)}`,
  { label: 'assemble', phase: 'Assemble', schema: GUIDE_SCHEMA })
// stamp provenance.coverage deterministically (do not trust the model for the count)
assembled.module = CATEGORY
assembled.version = assembled.version || '1.0'
assembled.system_module = CATEGORY
assembled.rubric_version = assembled.rubric_version || (RUBRIC.version || '1.0')
const addressedCount = CRIT_IDS.length - finalUncovered.length
assembled.provenance = {
  sources: allTech.map(t => t.source).filter(Boolean),
  confidence: finalUncovered.length === 0 ? 'high' : 'medium',
  coverage: { criteria_total: CRIT_IDS.length, criteria_addressed: addressedCount, uncovered: finalUncovered },
}

// ---- Review: completeness ----
phase('Review')
const review = await agent(
  `Completeness review of the "${CATEGORY}" BUILD GUIDE. Killer systems that MUST appear as build steps: ${JSON.stringify(KILLER)}. Report which are covered vs missing, whether the build ORDER is sound (dependencies before dependents), and an overall assessment of whether this guide, if followed, produces an above-the-mean build that clears the rubric (coverage = ${addressedCount}/${CRIT_IDS.length}).\n\nGUIDE:\n${JSON.stringify(assembled.tiers.map(t => ({ tier: t.tier, steps: (t.steps || []).map(s => ({ id: s.id, instruction: s.instruction })) })), null, 1)}`,
  { label: 'review', phase: 'Review', schema: REVIEW_SCHEMA })

return {
  category: CATEGORY,
  guide: assembled,
  coverage: { total: CRIT_IDS.length, addressed: addressedCount, uncovered: finalUncovered },
  review,
  provenance: { angles: ANGLES.map(a => a.key), techniqueCount: allTech.length, fillRounds: round },
}
```
