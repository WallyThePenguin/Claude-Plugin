# SKILLS Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give the in-Studio agent **skills** — named, reusable recipes it invokes via `use_skill(name, args?)` to do common Roblox work well: built-in build recipes + Studio-adapted ports of Claude's process skills, plus self-authored (global-scope, persisted) skills.

**Architecture:** A skill is recipe TEXT, not an executor — `use_skill` returns the (parameter-resolved) body and the agent follows it with its existing tools (composes with `dispatch_writers`/`update_tasks`/`capture`). An always-on catalog (`name — description` lines) rides the leading-message preamble alongside `SystemPrompt` + memory (the proxy strips `system`). Built-ins are code modules; user skills persist via `plugin:SetSetting`, mirroring `Memory.luau`. Three parent-only, non-destructive tools (`use_skill`/`save_skill`/`delete_skill`).

**Tech Stack:** Roblox Luau plugin; Rojo build; `selene` lint; `stylua` format. Spec: `docs/superpowers/specs/2026-06-17-studio-agent-skills-design.md`.

**Testing note (same as prior features):** Luau using Roblox APIs can't run outside Studio. The PURE layer (`SkillCatalog`) gets an in-Studio self-test (`SkillCatalogSpec`, run by temporarily adding `require(script.SkillCatalogSpec).run()` to `init.server.luau`). Per-task verification is the static gate — `stylua --check src`, `selene src` (0 errors; the 1 known `CreateDockWidgetPluginGui` warning), `rojo build default.project.json -o StudioAgent.rbxm`. Real behavioral verification = the adversarial-review Workflow + in-Studio live-verify in Task 9.

**Commands (PowerShell, repo root):** `stylua src` (format) · `stylua --check src` · `selene src` · `rojo build default.project.json -o StudioAgent.rbxm` · `.\build.ps1` (build + install).

Create `src/SkillCatalog.luau`:

```lua
--!strict
-- SkillCatalog: PURE helpers for skills (no self, no DataModel — directly unit-testable, the bulk of
-- the logic; mirrors Memory.luau's pure layer). A skill is recipe TEXT the agent executes with its
-- existing tools; use_skill returns the rendered body. Built-ins are code modules; user skills persist
-- (see SkillStore). normalizeName/validateSkill/renderSkill/catalogBlock are shared by SkillStore +
-- Tools/Skills + the Spec.

local SkillCatalog = {}

export type Param = { name: string, description: string, required: boolean }
export type Skill = {
	name: string,
	description: string,
	body: string,
	parameters: { Param },
	category: string, -- "recipe" | "working"
	source: string, -- "builtin" | "user"
}

local MAX_NAME = 48
local MAX_DESC = 200
local MAX_BODY = 6000
local MAX_PARAMS = 8
local MAX_PARAM_FIELD = 200
local MAX_USER_SKILLS = 20

SkillCatalog.MAX_NAME = MAX_NAME
SkillCatalog.MAX_DESC = MAX_DESC
SkillCatalog.MAX_BODY = MAX_BODY
SkillCatalog.MAX_PARAMS = MAX_PARAMS
SkillCatalog.MAX_PARAM_FIELD = MAX_PARAM_FIELD
SkillCatalog.MAX_USER_SKILLS = MAX_USER_SKILLS

-- ---- shared string helpers (UTF-8-safe; same shape as Memory.luau) ----

local function trim(s: string): string
	return (string.gsub(s, "^%s*(.-)%s*$", "%1"))
end

local function truncate(s: string, maxChars: number): string
	local len = utf8.len(s)
	if len == nil then
		return string.sub(s, 1, maxChars)
	end
	if len <= maxChars then
		return s
	end
	local byteEnd = utf8.offset(s, maxChars + 1)
	if byteEnd == nil then
		return s
	end
	return string.sub(s, 1, byteEnd - 1)
end

-- Replace EVERY literal occurrence of `find` in `s` with `repl` (NOT a Lua pattern — escapes magic
-- chars, so a param name like "shop-gui" with a magic '-' is replaced literally).
local function literalReplace(s: string, find: string, repl: string): string
	local escaped = (string.gsub(find, "[%(%)%.%%%+%-%*%?%[%]%^%$]", "%%%1"))
	return (string.gsub(s, escaped, (string.gsub(repl, "%%", "%%%%"))))
end

-- ---- name / param / skill normalization ----

-- Lowercase, trim, collapse non-[a-z0-9] runs to "-", strip leading/trailing "-"; nil if empty.
function SkillCatalog.normalizeName(name: any): string?
	if type(name) ~= "string" then
		return nil
	end
	local s = string.lower(trim(name))
	s = (string.gsub(s, "[^a-z0-9]+", "-"))
	s = (string.gsub(s, "^%-+", ""))
	s = (string.gsub(s, "%-+$", ""))
	if s == "" then
		return nil
	end
	return truncate(s, MAX_NAME)
end

-- Defensive: drop malformed entries, trim+truncate fields, coerce required to boolean, cap count.
function SkillCatalog.normalizeParams(raw: any): { Param }
	local out: { Param } = {}
	if type(raw) ~= "table" then
		return out
	end
	for _, item in ipairs(raw :: { any }) do
		if type(item) == "table" and type(item.name) == "string" and trim(item.name) ~= "" then
			local desc = if type(item.description) == "string" then item.description else ""
			table.insert(out, {
				name = truncate(trim(item.name), MAX_PARAM_FIELD),
				description = truncate(trim(desc), MAX_PARAM_FIELD),
				required = item.required == true,
			})
			if #out >= MAX_PARAMS then
				break
			end
		end
	end
	return out
end

-- Validate+normalize a raw skill table into a clean Skill, or (nil, errorString).
function SkillCatalog.validateSkill(raw: any, source: string, category: string): (Skill?, string?)
	if type(raw) ~= "table" then
		return nil, "skill must be a table"
	end
	local name = SkillCatalog.normalizeName(raw.name)
	if name == nil then
		return nil, "skill name must contain at least one letter/digit"
	end
	if type(raw.description) ~= "string" or trim(raw.description) == "" then
		return nil, "skill '" .. name .. "' needs a non-empty description"
	end
	if type(raw.body) ~= "string" or trim(raw.body) == "" then
		return nil, "skill '" .. name .. "' needs a non-empty body"
	end
	return {
		name = name,
		description = truncate(trim(raw.description), MAX_DESC),
		body = truncate(trim(raw.body), MAX_BODY),
		parameters = SkillCatalog.normalizeParams(raw.parameters),
		category = category,
		source = source,
	},
		nil
end

-- Exact (normalized) name lookup.
function SkillCatalog.findByName(skills: { Skill }, name: any): Skill?
	local target = SkillCatalog.normalizeName(name)
	if target == nil then
		return nil
	end
	for _, s in ipairs(skills) do
		if s.name == target then
			return s
		end
	end
	return nil
end

-- Render a skill into the text use_skill returns. Required-param missing -> (errorText, false) BEFORE
-- any build. Substitutes {{param}} placeholders, then appends a Provided-arguments block.
function SkillCatalog.renderSkill(skill: Skill, args: any): (string, boolean)
	local a: { [string]: any } = if type(args) == "table" then args else {}
	local missing: { string } = {}
	for _, p in ipairs(skill.parameters) do
		if p.required and (a[p.name] == nil or a[p.name] == "") then
			table.insert(missing, p.name)
		end
	end
	if #missing > 0 then
		return "skill '"
			.. skill.name
			.. "' needs argument(s): "
			.. table.concat(missing, ", ")
			.. ". Call use_skill again with them in `args`.",
			false
	end
	local body = skill.body
	for _, p in ipairs(skill.parameters) do
		if a[p.name] ~= nil then
			body = literalReplace(body, "{{" .. p.name .. "}}", tostring(a[p.name]))
		end
	end
	local lines = { "# Skill: " .. skill.name, skill.description, "", body }
	if #skill.parameters > 0 then
		table.insert(lines, "")
		table.insert(lines, "## Provided arguments")
		for _, p in ipairs(skill.parameters) do
			local v = if a[p.name] ~= nil then tostring(a[p.name]) else "(not provided)"
			table.insert(lines, "- " .. p.name .. ": " .. v)
		end
	end
	return table.concat(lines, "\n"), true
end

-- Render the always-on catalog (name — description lines, grouped). "" if no skills.
function SkillCatalog.catalogBlock(skills: { Skill }): string
	local recipes: { Skill } = {}
	local working: { Skill } = {}
	local user: { Skill } = {}
	for _, s in ipairs(skills) do
		if s.source == "user" then
			table.insert(user, s)
		elseif s.category == "working" then
			table.insert(working, s)
		else
			table.insert(recipes, s)
		end
	end
	if #recipes == 0 and #working == 0 and #user == 0 then
		return ""
	end
	local out = {
		"# Skills (named recipes you can invoke)",
		"Call use_skill(name, args?) to load a recipe, then FOLLOW it with your normal tools — it returns "
			.. "instructions, not a finished build.",
	}
	local function section(title: string, list: { Skill })
		if #list == 0 then
			return
		end
		table.insert(out, "")
		table.insert(out, "## " .. title)
		for _, s in ipairs(list) do
			table.insert(out, "- " .. s.name .. " — " .. s.description)
		end
	end
	section("Build recipes", recipes)
	section("Working skills (how to approach a task)", working)
	section("Your skills (saved)", user)
	return table.concat(out, "\n")
end

return SkillCatalog
```

Create `src/SkillCatalogSpec.luau`:

```lua
--!strict
-- SkillCatalogSpec: in-Studio self-test for the PURE SkillCatalog helpers (no DataModel). Run via
-- temporarily adding `require(script.SkillCatalogSpec).run()` to init.server.luau, build, reopen
-- Studio, read Output.

local SkillCatalog = require(script.Parent.SkillCatalog)

local M = {}

type Case = { name: string, fn: () -> (boolean, string?) }

local function buildCases(): { Case }
	local cases: { Case } = {}
	local function add(name: string, fn: () -> (boolean, string?))
		table.insert(cases, { name = name, fn = fn })
	end

	add("normalizeName slugifies and rejects empty", function()
		return SkillCatalog.normalizeName("Shop GUI!") == "shop-gui"
			and SkillCatalog.normalizeName("  --R15 NPC--  ") == "r15-npc"
			and SkillCatalog.normalizeName("") == nil
			and SkillCatalog.normalizeName("!!!") == nil
			and SkillCatalog.normalizeName(5) == nil
	end)

	add("normalizeParams drops malformed, caps count, coerces required", function()
		local raw = {
			{ name = "title", description = "the title", required = true },
			{ name = "  ", description = "blank name" },
			{ name = "count" },
			"nope",
		}
		local p = SkillCatalog.normalizeParams(raw)
		return #p == 2
			and p[1].name == "title"
			and p[1].required == true
			and p[2].name == "count"
			and p[2].required == false
	end)

	add("validateSkill accepts a good skill and normalizes the name", function()
		local s, err = SkillCatalog.validateSkill(
			{ name = "Shop GUI", description = " build a shop ", body = " do it ", parameters = {} },
			"builtin",
			"recipe"
		)
		return err == nil
			and s ~= nil
			and s.name == "shop-gui"
			and s.description == "build a shop"
			and s.body == "do it"
			and s.category == "recipe"
			and s.source == "builtin"
	end)

	add("validateSkill rejects bad name / empty description / empty body", function()
		local _, e1 = SkillCatalog.validateSkill({ name = "!!!", description = "x", body = "y" }, "user", "recipe")
		local _, e2 = SkillCatalog.validateSkill({ name = "ok", description = "  ", body = "y" }, "user", "recipe")
		local _, e3 = SkillCatalog.validateSkill({ name = "ok", description = "x", body = "  " }, "user", "recipe")
		return e1 ~= nil and e2 ~= nil and e3 ~= nil
	end)

	add("renderSkill errors on a missing required param before building", function()
		local skill = (SkillCatalog.validateSkill({
			name = "x",
			description = "d",
			body = "build {{title}}",
			parameters = { { name = "title", description = "t", required = true } },
		}, "builtin", "recipe"))
		local text, ok = SkillCatalog.renderSkill(skill :: any, {})
		return ok == false and string.find(text, "title", 1, true) ~= nil
	end)

	add("renderSkill substitutes {{param}} (incl. a hyphenated name) and appends args block", function()
		local skill = (SkillCatalog.validateSkill({
			name = "x",
			description = "d",
			body = "make {{item-name}} now",
			parameters = { { name = "item-name", description = "n", required = false } },
		}, "builtin", "recipe"))
		local text, ok = SkillCatalog.renderSkill(skill :: any, { ["item-name"] = "Sword" })
		return ok == true
			and string.find(text, "make Sword now", 1, true) ~= nil
			and string.find(text, "## Provided arguments", 1, true) ~= nil
			and string.find(text, "- item-name: Sword", 1, true) ~= nil
	end)

	add("renderSkill leaves an undeclared placeholder literal", function()
		local skill = (SkillCatalog.validateSkill({ name = "x", description = "d", body = "{{nope}}" }, "builtin", "recipe"))
		local text = SkillCatalog.renderSkill(skill :: any, {})
		return string.find(text, "{{nope}}", 1, true) ~= nil
	end)

	add("findByName matches on the normalized name", function()
		local s = (SkillCatalog.validateSkill({ name = "shop-gui", description = "d", body = "b" }, "builtin", "recipe"))
		local list = { s :: any }
		return SkillCatalog.findByName(list, "Shop GUI") ~= nil and SkillCatalog.findByName(list, "missing") == nil
	end)

	add("catalogBlock groups by family and is empty when no skills", function()
		if SkillCatalog.catalogBlock({}) ~= "" then
			return false, "empty not blank"
		end
		local recipe = (SkillCatalog.validateSkill({ name = "leaderstats", description = "stats", body = "b" }, "builtin", "recipe"))
		local work = (SkillCatalog.validateSkill({ name = "test-first", description = "tdd", body = "b" }, "builtin", "working"))
		local mine = (SkillCatalog.validateSkill({ name = "my-thing", description = "mine", body = "b" }, "user", "recipe"))
		local txt = SkillCatalog.catalogBlock({ recipe :: any, work :: any, mine :: any })
		return string.find(txt, "## Build recipes", 1, true) ~= nil
			and string.find(txt, "## Working skills", 1, true) ~= nil
			and string.find(txt, "## Your skills", 1, true) ~= nil
			and string.find(txt, "- leaderstats — stats", 1, true) ~= nil
	end)

	return cases
end

function M.run(): { passed: boolean, total: number, failed: number }
	local cases = buildCases()
	local total = #cases
	local failed = 0
	print("=== SkillCatalogSpec: running " .. tostring(total) .. " cases ===")
	for _, case in ipairs(cases) do
		local ok, result, detail = pcall(case.fn)
		local pass = ok and result == true
		if pass then
			print("  PASS  " .. case.name)
		else
			failed += 1
			local why = if not ok then tostring(result) else tostring(detail)
			print("  FAIL  " .. case.name .. (if why ~= "nil" then " (" .. why .. ")" else ""))
		end
	end
	local passed = total - failed
	print("=== Summary: " .. tostring(passed) .. "/" .. tostring(total) .. " passed, " .. tostring(failed) .. " failed ===")
	return { passed = failed == 0, total = total, failed = failed }
end

return M
```

Create `src/SkillStore.luau`:

```lua
--!strict
-- SkillStore: the live skills store. Merges BUILT-IN skills (code modules, immutable) with USER skills
-- (self-authored, persisted via plugin:SetSetting — same mechanism as Memory). One instance in
-- deps.skills. Built-in names are RESERVED (save rejects a collision; delete refuses a built-in), so
-- the merged catalog has unique names and use_skill is unambiguous. User skills are GLOBAL-scope (one
-- bucket) in v1.

local SkillCatalog = require(script.Parent.SkillCatalog)
local BuiltinRecipes = require(script.Parent.BuiltinRecipes)
local BuiltinWorkingSkills = require(script.Parent.BuiltinWorkingSkills)

type Skill = SkillCatalog.Skill

local SkillStore = {}
SkillStore.__index = SkillStore

local SETTING_PREFIX = "StudioAgent_"
local USER_KEY = SETTING_PREFIX .. "skills_user"

export type SkillStore = typeof(setmetatable(
	{} :: {
		_plugin: Plugin,
		_builtins: { Skill },
		_builtinNames: { [string]: boolean },
	},
	SkillStore
))

-- Validate the built-in lists ONCE at construction; a malformed built-in is dropped + warned (never
-- crashes the plugin).
local function buildBuiltins(): ({ Skill }, { [string]: boolean })
	local out: { Skill } = {}
	local names: { [string]: boolean } = {}
	local function add(rawList: { any }, category: string)
		for _, raw in ipairs(rawList) do
			local skill, err = SkillCatalog.validateSkill(raw, "builtin", category)
			if skill ~= nil and not names[skill.name] then
				names[skill.name] = true
				table.insert(out, skill)
			elseif skill == nil then
				warn("[StudioAgent] skipping malformed built-in skill: " .. tostring(err))
			end
		end
	end
	add(BuiltinRecipes.list, "recipe")
	add(BuiltinWorkingSkills.list, "working")
	return out, names
end

function SkillStore.new(plugin: Plugin): SkillStore
	local builtins, names = buildBuiltins()
	return setmetatable({ _plugin = plugin, _builtins = builtins, _builtinNames = names }, SkillStore)
end

-- Load user skills, re-validating defensively (a corrupt/old blob can't inject junk); drop a name that
-- collides with a built-in or a duplicate; cap to MAX_USER_SKILLS (keep the newest).
function SkillStore.loadUser(self: SkillStore): { Skill }
	local raw = self._plugin:GetSetting(USER_KEY)
	local out: { Skill } = {}
	if type(raw) == "table" then
		local seen: { [string]: boolean } = {}
		for _, e in ipairs(raw :: { any }) do
			local skill = SkillCatalog.validateSkill(e, "user", "recipe")
			if skill ~= nil and not self._builtinNames[skill.name] and not seen[skill.name] then
				seen[skill.name] = true
				table.insert(out, skill)
			end
		end
	end
	while #out > SkillCatalog.MAX_USER_SKILLS do
		table.remove(out, 1)
	end
	return out
end

-- Built-ins first, then user. Used by the catalog + lookup.
function SkillStore.all(self: SkillStore): { Skill }
	local out: { Skill } = {}
	for _, s in ipairs(self._builtins) do
		table.insert(out, s)
	end
	for _, s in ipairs(self:loadUser()) do
		table.insert(out, s)
	end
	return out
end

-- Persist the raw user list (store only the authored fields).
local function persistUser(self: SkillStore, skills: { Skill })
	local raw: { any } = {}
	for _, s in ipairs(skills) do
		table.insert(raw, { name = s.name, description = s.description, body = s.body, parameters = s.parameters })
	end
	self._plugin:SetSetting(USER_KEY, raw)
end

-- Save a user skill. Rejects a built-in-name collision; updates an existing same-name user skill in
-- place; FIFO-drops the oldest if over cap. Returns (note, ok).
function SkillStore.save(self: SkillStore, name: any, description: any, body: any, parameters: any): (string, boolean)
	local skill, err = SkillCatalog.validateSkill(
		{ name = name, description = description, body = body, parameters = parameters },
		"user",
		"recipe"
	)
	if skill == nil then
		return "save_skill failed: " .. tostring(err), false
	end
	if self._builtinNames[skill.name] then
		return "save_skill failed: '" .. skill.name .. "' is a built-in skill name (pick another).", false
	end
	local users = self:loadUser()
	local replaced = false
	for i, s in ipairs(users) do
		if s.name == skill.name then
			users[i] = skill
			replaced = true
			break
		end
	end
	if not replaced then
		table.insert(users, skill)
	end
	local dropped = 0
	while #users > SkillCatalog.MAX_USER_SKILLS do
		table.remove(users, 1)
		dropped += 1
	end
	persistUser(self, users)
	local note = if replaced then "updated skill '" .. skill.name .. "'" else "saved skill '" .. skill.name .. "'"
	note ..= " (" .. tostring(#users) .. " of your skills stored)"
	if dropped > 0 then
		note ..= "; dropped " .. tostring(dropped) .. " oldest (cap " .. tostring(SkillCatalog.MAX_USER_SKILLS) .. ")"
	end
	return note, true
end

-- Delete a USER skill by name. Refuses a built-in. Returns (note, ok).
function SkillStore.delete(self: SkillStore, name: any): (string, boolean)
	local n = SkillCatalog.normalizeName(name)
	if n == nil then
		return "delete_skill: invalid name", false
	end
	if self._builtinNames[n] then
		return "delete_skill: '" .. n .. "' is a built-in skill and can't be deleted.", false
	end
	local users = self:loadUser()
	local kept: { Skill } = {}
	local removed = false
	for _, s in ipairs(users) do
		if s.name == n then
			removed = true
		else
			table.insert(kept, s)
		end
	end
	if not removed then
		return "delete_skill: no skill named '" .. n .. "'", false
	end
	persistUser(self, kept)
	return "deleted skill '" .. n .. "'", true
end

-- Resolve name+args into the recipe text use_skill returns. (text, ok).
function SkillStore.resolve(self: SkillStore, name: any, args: any): (string, boolean)
	local skill = SkillCatalog.findByName(self:all(), name)
	if skill == nil then
		return "unknown skill '" .. tostring(name) .. "'. See the skill catalog for valid names.", false
	end
	return SkillCatalog.renderSkill(skill, args)
end

function SkillStore.listBuiltins(self: SkillStore): { Skill }
	return self._builtins
end

function SkillStore.listUser(self: SkillStore): { Skill }
	return self:loadUser()
end

-- The always-on catalog injected into the leading preamble.
function SkillStore.catalogBlock(self: SkillStore): string
	return SkillCatalog.catalogBlock(self:all())
end

return SkillStore
```
